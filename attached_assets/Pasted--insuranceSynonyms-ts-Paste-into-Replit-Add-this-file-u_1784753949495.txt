const normalise = (value: string) =>
  value
    .toLowerCase()
    .replace(/&/g, ' and ')
    .replace(/[.,/()\-–—]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();

const synonymIndex = new Map<string, SynonymEntry>();

for (const entry of INSURANCE_SYNONYMS) {
  synonymIndex.set(normalise(entry.canonical), entry);
  for (const alias of entry.aliases) {
    synonymIndex.set(normalise(alias), entry);
  }
}

export function normaliseInsuranceTerm(term: string): string {
  const entry = synonymIndex.get(normalise(term));
  return entry ? entry.canonical : term;
}

export function getInsuranceSynonymEntry(term: string): SynonymEntry | null {
  return synonymIndex.get(normalise(term)) ?? null;
}

export function extractInsuranceMatches(text: string): SynonymEntry[] {
  const matches = new Map<string, SynonymEntry>();
  const haystack = normalise(text);

  for (const entry of INSURANCE_SYNONYMS) {
    const allTerms = [entry.canonical, ...entry.aliases];
    for (const term of allTerms) {
      const needle = normalise(term);
      if (needle && haystack.includes(needle)) {
        matches.set(entry.canonical, entry);
        break;
      }
    }
  }

  return [...matches.values()].sort((a, b) => (b.priority ?? 0) - (a.priority ?? 0));
}

export function replaceInsuranceTerms(text: string): string {
  let output = text;
  const allTerms = INSURANCE_SYNONYMS
    .flatMap((entry) => [entry.canonical, ...entry.aliases])
    .sort((a, b) => b.length - a.length);

  for (const term of allTerms) {
    const canonical = normaliseInsuranceTerm(term);
    const escaped = term.replace(/[.*+?^${}()|[\]\\]/g, '\\$&').replace(/\\&/g, 'and');
    const pattern = new RegExp(`\\b${escaped}\\b`, 'gi');
    output = output.replace(pattern, canonical);
  }

  return output;
}

export function buildInsuranceHintBlock(): string {
  const grouped = INSURANCE_SYNONYMS.reduce<Record<string, string[]>>((acc, entry) => {
    if (!acc[entry.kind]) acc[entry.kind] = [];
    acc[entry.kind].push(`${entry.canonical}: ${entry.aliases.join(', ')}`);
    return acc;
  }, {});

  return Object.entries(grouped)
    .map(([kind, items]) => `${kind.toUpperCase()}\n- ${items.join('\n- ')}`)
    .join('\n\n');
}

export function debugInsuranceSynonyms() {
  return {
    totalEntries: INSURANCE_SYNONYMS.length,
    sample: INSURANCE_SYNONYMS.slice(0, 5),
    hintBlock: buildInsuranceHintBlock(),
  };
}
